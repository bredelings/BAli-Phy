# The `--set` command:

--set <key>=<value>                         Set configuration value.

Pass a key-value pair into BAli-Phy.  The <value> is a json value.

# Examples:

   # Work twice at hard at resampling the alignment (default: 1.0)
   bali-phy --set alignment-sampling-factor=2.0 dna.fasta

   # Write the alignment every 10th logged iteration (default: 10)
   bali-phy --set alignment-extra-subsample=10 dna.fasta

   # Infer ambiguous characters for leaf sequences (default: false)
   bali-phy --set infer-ambiguous-observed=true dna.fasta -I none

   # Write fixed alignments (useful for ancestral sequence reconstruction) (default: false)
   bali-phy --set write-fixed-alignments=true dna.fasta -I none

   # Write property values for each (category, state) pair and also the
   # (category,state) pair for each leaf character. Format is JSON. (default: true)
   bali-phy --set write-properties=true dna.fasta

   # Stop enforcing low gap probabilities after 10 iterations (default: 10)
   bali-phy --set alignment-burnin=10 dna.fasta

   # Disallow site-compression (default: true)
   bali-phy --set site-compression=false dna.fasta -I none
